function shuffle(a){
  const arr=[...a];
  for(let i=arr.length-1;i>0;i--){
    const buf=new Uint32Array(1);
    crypto.getRandomValues(buf);
    const j=buf[0]%(i+1);
    [arr[i],arr[j]]=[arr[j],arr[i]];
  }
  return arr;
}

function nextPairReady(){
  while(!state.finished){
    if(state.current.length===1){
      state.finished=true;state.champion=state.current[0];save();showResult();return false;
    }
    if(state.index >= state.current.length){
      state.current=state.winners;
      state.winners=[];
      state.index=0;
      state.round++;
      if(state.current.length===1){
        state.finished=true;state.champion=state.current[0];save();showResult();return false;
      }
      continue;
    }
    if(state.index===state.current.length-1){
      state.winners.push(state.current[state.index]);
      state.index++;
      continue;
    }
    return true;
  }
  return false;
}

const PRELOADED_POSTERS = new Set();

function proxiedPosterUrl(src, mode="thumb"){
  if(!src) return null;
  const mobile = window.matchMedia && window.matchMedia("(max-width: 700px)").matches;
  if(mode==="thumb"){
    const w = mobile ? 240 : 320;
    const h = Math.round(w * 1.5);
    return `https://wsrv.nl/?url=${encodeURIComponent(src)}&w=${w}&h=${h}&fit=cover&output=webp&q=48`;
  }
  const w = mobile ? 420 : 560;
  const h = Math.round(w * 1.5);
  return `https://wsrv.nl/?url=${encodeURIComponent(src)}&w=${w}&h=${h}&fit=cover&output=webp&q=72`;
}

function originalPosterSources(m){
  return [...new Set([
    ...(DIRECT_POSTERS[m.id] || []),
    m.poster,
    m.douban ? `https://dou.img.lithub.cc/movie/${m.douban}.jpg` : null
  ].filter(Boolean))];
}

function thumbCandidates(m){
  return originalPosterSources(m).map(src => proxiedPosterUrl(src,"thumb")).filter(Boolean);
}

function shouldUpgradePoster(){
  const c = navigator.connection || navigator.mozConnection || navigator.webkitConnection;
  if(!c) return true;
  if(c.saveData) return false;
  return !["slow-2g","2g"].includes(c.effectiveType);
}

function preloadMoviePoster(m){
  if(!m || PRELOADED_POSTERS.has(m.id)) return;
  PRELOADED_POSTERS.add(m.id);
  const src = thumbCandidates(m)[0];
  if(!src) return;
  const pre = new Image();
  pre.decoding = "async";
  pre.fetchPriority = "low";
  pre.src = src;
}
