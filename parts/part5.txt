}

function posterCandidates(m){
  const originals = [
    ...(DIRECT_POSTERS[m.id] || []),
    m.poster,
    m.douban ? `https://dou.img.lithub.cc/movie/${m.douban}.jpg` : null
  ].filter(Boolean);
  const unique = [...new Set(originals)];
  // Try lightweight proxied image first, then original as fallback.
  return unique.flatMap(src => [fastPosterUrl(src), src]).filter(Boolean);
}

function resolvePoster(m,img){
  const wrap = img.parentElement;
  const ph = wrap ? wrap.querySelector(".poster-placeholder") : null;

  const candidates = posterCandidates(m);
  let i = 0;

  img.style.display = "block";
  img.style.opacity = "1";
  img.style.visibility = "visible";

  const reveal = () => {
    img.style.display = "block";
    img.style.opacity = "1";
    img.style.visibility = "visible";
    img.style.position = "absolute";
    img.style.inset = "0";
    img.style.zIndex = "1";
    if(ph) ph.style.display = "none";
  };

  const tryNext = () => {
    if(i >= candidates.length){
      posterFallback(wrap,m);
      return;
    }
    const src = candidates[i++];
    img.onload = () => {
      reveal();
      requestAnimationFrame(() => reveal());
      setTimeout(reveal, 0);
    };
    img.onerror = () => tryNext();
    img.src = src;
  };
  tryNext();
}

/* Mobile-first poster loader v8: show a tiny WebP first, then upgrade quietly. */
function fastPosterUrl(src){
  if(!src) return null;
  const mobile = window.matchMedia && window.matchMedia("(max-width: 700px)").matches;
  const w = mobile ? 240 : 320;
  const h = Math.round(w * 1.5);
  return `https://wsrv.nl/?url=${encodeURIComponent(src)}&w=${w}&h=${h}&fit=cover&output=webp&q=48`;
}
function sharpPosterUrl(src){
  if(!src) return null;
  const mobile = window.matchMedia && window.matchMedia("(max-width: 700px)").matches;
  const w = mobile ? 420 : 560;
  const h = Math.round(w * 1.5);
  return `https://wsrv.nl/?url=${encodeURIComponent(src)}&w=${w}&h=${h}&fit=cover&output=webp&q=72`;
}
function originalPosterSources(m){
  return [...new Set([
    ...(DIRECT_POSTERS[m.id] || []),
    m.poster,
    m.douban ? `https://dou.img.lithub.cc/movie/${m.douban}.jpg` : null
  ].filter(Boolean))];
}
function posterCandidates(m){
  return originalPosterSources(m).map(fastPosterUrl).filter(Boolean);
}
function shouldUpgradePoster(){
  const c = navigator.connection || navigator.mozConnection || navigator.webkitConnection;
  if(!c) return true;
  if(c.saveData) return false;
  return !["slow-2g","2g"].includes(c.effectiveType);
}

}

function resolvePoster(m,img){
  const wrap = img.parentElement;
  const ph = wrap ? wrap.querySelector(".poster-placeholder") : null;
  const originals = originalPosterSources(m);
  const thumbs = originals.map(fastPosterUrl);
  let i = 0;

  img.style.display = "block";
  img.style.opacity = "1";
  img.style.visibility = "visible";

  const reveal = () => {
    img.style.display = "block";
    img.style.opacity = "1";
    img.style.visibility = "visible";
    img.style.position = "absolute";
    img.style.inset = "0";
    img.style.zIndex = "1";
    if(ph) ph.style.display = "none";
  };

  const upgradeQuietly = (originalSrc) => {
    if(!shouldUpgradePoster() || !originalSrc) return;
    const sharp = new Image();
    sharp.decoding = "async";
    sharp.fetchPriority = "low";
    sharp.onload = () => {
      img.src = sharp.src;
      reveal();
    };
    sharp.src = sharpPosterUrl(originalSrc);
  };

  const tryThumb = () => {
    if(i >= thumbs.length){
      let j=0;
      const tryOriginal = () => {
        if(j >= originals.length){ posterFallback(wrap,m); return; }
        const src = originals[j++];
        img.onload = reveal;
        img.onerror = tryOriginal;
        img.src = src;
      };
      tryOriginal();
      return;
    }
    const sourceIndex = i;
    const src = thumbs[i++];
    img.onload = () => {
      reveal();
      setTimeout(() => upgradeQuietly(originals[sourceIndex]), 120);
    };
    img.onerror = tryThumb;
    img.src = src;
  };
  tryThumb();
}

function choose(side){
  if(state.finished || !nextPairReady()) return;
  const left=state.current[state.index], right=state.current[state.index+1];
  const winner=side==="left"?left:right;
  const loser=side==="left"?right:left;
  if(!state.history[winner]) state.history[winner]=[];
  state.history[winner].push(loser);
  state.winners.push(winner);
  state.index+=2;
  state.completed++;
  save();
  const chosen=side==="left"?leftCard:rightCard;
  chosen.style.borderColor="rgba(243,200,106,.88)";
  document.getElementById("arena").classList.add("fade-out");
  setTimeout(()=>{
    document.getElementById("arena").classList.remove("fade-out");
    chosen.style.borderColor="";
    render();
  },155);
}
