function shuffle(a){
  const arr=[...a];
  for(let i=arr.length-1;i>0;i--){
    const buf=new Uint32Array(1);
    crypto.getRandomValues(buf);
    const j=buf[0]%(i+1);
    [arr[i],arr[j]]=[arr[j],arr[i]];
  }
  return arr;
}

function nextPairReady(){
  while(!state.finished){
    if(state.current.length===1){
      state.finished=true;state.champion=state.current[0];save();showResult();return false;
    }
    if(state.index >= state.current.length){
      state.current=state.winners;
      state.winners=[];
      state.index=0;
      state.round++;
      if(state.current.length===1){
        state.finished=true;state.champion=state.current[0];save();showResult();return false;
      }
      continue;
    }
    if(state.index===state.current.length-1){
      state.winners.push(state.current[state.index]);
      state.index++;
      continue;
    }
    return true;
  }
  return false;
}
