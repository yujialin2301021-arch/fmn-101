dex = i;
    const src = thumbs[i++];
    img.onload = () => {
      reveal();
      // The usable thumbnail is already visible. Upgrade only in the background.

      setTimeout(() => upgradeQuietly(chosenOriginal), 80);

function render(){
  if(!nextPairReady()) return;
  game.style.display="block";result.classList.remove("show");
  const a=byId[state.current[state.index]], b=byId[state.current[state.index+1]];
  leftCard.innerHTML=cardHTML(a,"left");
  rightCard.innerHTML=cardHTML(b,"right");
  resolvePoster(a,leftCard.querySelector(".poster"));
  resolvePoster(b,rightCard.querySelector(".poster"));
  const roundMatches=Math.floor(state.current.length/2);
  const matchup=Math.floor(state.index/2)+1;
  progressText.textContent=`第 ${state.round} 轮 · 对决 ${matchup}/${roundMatches}`;
  countText.textContent=`已做出 ${state.completed}/57 次选择`;
  progressBar.style.width=`${Math.min(100,state.completed/57*100)}%`;
  roundChip.textContent=`第 ${state.round} 轮`;
  preloadUpcoming();
}
function choose(side){
  if(state.finished || !nextPairReady()) return;
  const left=state.current[state.index], right=state.current[state.index+1];
  const winner=side==="left"?left:right;
  const loser=side==="left"?right:left;
  if(!state.history[winner]) state.history[winner]=[];
  state.history[winner].push(loser);
  state.winners.push(winner);
  state.index+=2;
  state.completed++;
  save();
  const chosen=side==="left"?leftCard:rightCard;
  chosen.style.borderColor="rgba(243,200,106,.88)";
  document.getElementById("arena").classList.add("fade-out");
  setTimeout(()=>{
    document.getElementById("arena").classList.remove("fade-out");
    chosen.style.borderColor="";
    render();
  },155);
}

window.addEventListener("keydown",e=>{
  if(state.finished)return;
  if(e.key.toLowerCase()==="a") choose("left");
  if(e.key.toLowerCase()==="d") choose("right");
});

}
